'use strict';
const el=id=>document.getElementById(id);
let assessmentText='', lastReport='', parsedAssessment=null;
const CORE_NAMES={"F01":"HR Governance & Policies","F02":"Workforce Planning & Manpower","F03":"Personnel & HR Operations","F04":"Compliance & HR Risk","F05":"Talent Acquisition & Onboarding","F06":"Performance Management","F07":"Learning & Development","F08":"Talent Management & Succession","F09":"Compensation & Benefits","F10":"Employee Relations & Engagement","F11":"HRIS, Data & Analytics","S01":"Business & HR Strategy Alignment","S02":"Organization Design & Structure","S03":"Workforce Capability","S04":"Leadership & Talent","S05":"Culture & Organizational Health","S06":"HR Operating Model","S07":"HR Technology & Digital Maturity","S08":"HR Analytics & Business Impact","S09":"Change & Transformation Readiness"};
function localReport(a) {
 const scope=a.scope||{mode:'full'},vals=a.answers||{};
 const ids=Object.keys(CORE_NAMES);
 const chosen=scope.mode==='functional'?ids.filter(x=>x.startsWith('F')):scope.mode==='focused'?
   (Array.isArray(scope.selected)?scope.selected.filter(x=>ids.includes(x)):[]):ids;
 const summary=chosen.map(id=>{
   const keys=Object.keys(vals).filter(k=>k.startsWith('SCA-'+id+'-Q'));
   const yes=keys.filter(k=>vals[k]&&vals[k].value===1).length;
   const no=keys.filter(k=>vals[k]&&vals[k].value===0).length;
   return{id,yes,no,complete:yes+no,name:CORE_NAMES[id]};
 });
 const answered=summary.reduce((n,c)=>n+c.complete,0),no=summary.reduce((n,c)=>n+c.no,0);
 const lines=[
 'WINNERS × SCA — تقرير مبدئي فوري (بدون Gemini)',
 'هذه نتائج أولية مبنية على إجابات المُقيّم؛ لا تثبت التطبيق أو القصور ولا تُعد رأيًا قانونيًا.',
 'النطاق: '+(scope.mode||'full')+' | المحاور: '+summary.length+' | الإجابات: '+answered+' | لا: '+no,
 '',
 'أولًا: ملخص إجابات المحاور'
 ];
 summary.forEach(c=>lines.push('• '+c.id+' '+c.name+': نعم '+c.yes+' / لا '+c.no+' / بدون إجابة '+(5-c.complete)));
 lines.push('','ثانيًا: أولويات التحقق (ليست فجوات مثبتة)');
 const gaps=summary.filter(c=>c.no>0).sort((a,b)=>b.no-a.no);
 if(gaps.length)gaps.forEach(c=>lines.push('• '+c.id+' '+c.name+' — يلزم التحقق من الأدلة المعتمدة قبل الحكم ('+c.no+' إجابات لا).'));
 else lines.push('• لا توجد إجابات «لا» مرصودة بالنطاق الحالي، لكن لا تُعد هذه شهادة امتثال.');
 lines.push('','ثالثًا: خارطة تحقق استرشادية 30 / 60 / 90 يومًا',
 '30 يوم: مراجعة المستندات السارية والتفويضات والنطاق والمسؤوليات.',
 '60 يوم: التحقق الميداني من الإجراءات والأدلة وإثبات الفجوات الفعلية عند وجودها.',
 '90 يوم: وضع إجراءات تحسين معتمدة وإعادة القياس بمؤشرات موثقة.',
 '','رابعًا: ضوابط الاستخدام',
 'هذا التقرير مبني على قواعد ثابتة وليس من إنتاج الذكاء الاصطناعي.',
 'لا يحتسب السؤال غير المُجاب أو المحور خارج النطاق صفرًا.',
 'لا تتغير درجات SCA Maturity الأصلية، وتظل النتائج خاضعة لمراجعة الاستشاري.');
 return lines.join('\n');
}

function msg(s,error){el('message').textContent=s;el('message').classList.toggle('error',!!error);}
function busy(v){el('testBtn').disabled=v;el('modelsBtn').disabled=v;el('generationBtn').disabled=v;el('analyzeBtn').disabled=v||!assessmentText||!el('approval').checked;}
function init(){google.script.run.withSuccessHandler(s=>{
  el('keyStatus').textContent=s.hasKey?'مضاف — جاهز للاختبار':'لم يُضف بعد';
  el('modelStatus').textContent=s.model;
  el('researchStatus').textContent=s.researchContextEnabled?'مفعّل بإذن المالك':'مُعطّل (آمن)';
}).withFailureHandler(e=>msg(e.message,true)).getBackendStatus();}
el('file').addEventListener('change',async e=>{
 const f=e.target.files[0];assessmentText='';parsedAssessment=null;el('analyzeBtn').disabled=true;el('offlineBtn').disabled=true;
 if(!f)return;
 if(f.size>120000){msg('الملف أكبر من الحد المسموح (120 KB).',true);return;}
 assessmentText=await f.text();
 try{const obj=JSON.parse(assessmentText);if(obj.schemaVersion!==1||obj.assessmentEdition!=='SCA-2026-11-10'||!obj.answers||typeof obj.answers!=='object')throw Error('not SCA');parsedAssessment=obj;
 msg('تم تحميل ملف SCA. لم يتم إرسال بيانات إلى Gemini بعد.');
 }catch(err){assessmentText='';msg('الملف ليس نسخة احتياطية صحيحة من SCA Maturity.',true);}
 el('analyzeBtn').disabled=!assessmentText||!el('approval').checked;el('offlineBtn').disabled=!assessmentText;
});
el('approval').onchange=()=>{el('analyzeBtn').disabled=!assessmentText||!el('approval').checked;};
el('offlineBtn').onclick=()=>{
 if(!parsedAssessment)return;
 lastReport=localReport(parsedAssessment);
 el('result').textContent=lastReport;el('saveBtn').disabled=false;
 msg('تم إعداد التقرير الأولي على الجهاز دون إرسال طلب إلى Gemini.');
};
el('testBtn').onclick=()=>{
 busy(true);msg('جاري فحص صلاحية المفتاح وقائمة نماذج Gemini (بدون توليد نص)...');
 let pending=true;
 const warning=setTimeout(()=>{
   if(pending)msg('اختبار الاتصال لم يرد خلال 20 ثانية. افتح Apps Script > Executions لمعرفة حالة testGeminiConnection، وراجع التفويض وصلاحية المفتاح.',true);
 },20000);
 google.script.run.withSuccessHandler(r=>{
   pending=false;clearTimeout(warning);busy(false);
   msg(r.ok?'نتيجة الاتصال ('+r.model+'): '+r.response:'تعذر التحقق من الاتصال.',!r.modelListed);
   init();
 }).withFailureHandler(e=>{
   pending=false;clearTimeout(warning);busy(false);
   msg('خطأ الاتصال: '+(e&&e.message?e.message:String(e)),true);
 }).testGeminiConnection();
};
el('modelsBtn').onclick=()=>{
 busy(true);msg('جاري فحص النماذج المرتبطة بالمفتاح...');
 google.script.run.withSuccessHandler(r=>{busy(false);msg('النماذج المتاحة: '+r.map(x=>x.name).join('، '));})
 .withFailureHandler(e=>{busy(false);msg(e.message,true);}).listGeminiModels();
};
el('generationBtn').onclick=()=>{
 busy(true);msg('جاري اختبار توليد نص صغير من Gemini، وقد يعيد المحرك المحاولة تلقائيًا عند 503...');
 let waiting=true;
 const warning=setTimeout(()=>{
   if(waiting)msg('الطلب لم ينته بعد. يتم انتظار استجابة Gemini أو محاولات إعادة محدودة. راجع Apps Script > Executions إذا استمر التأخير.',true);
 },20000);
 google.script.run.withSuccessHandler(r=>{
   waiting=false;clearTimeout(warning);busy(false);
   msg(r.ok?'نجح اختبار توليد النص ('+r.model+'): '+r.result:'لم ينتج Gemini نصًا.',!r.ok);
 }).withFailureHandler(e=>{
   waiting=false;clearTimeout(warning);busy(false);
   msg('تعذر اختبار التوليد: '+(e&&e.message?e.message:String(e)),true);
 }).testGeminiGeneration();
};
el('analyzeBtn').onclick=()=>{
 if(!parsedAssessment||!assessmentText||!el('approval').checked)return;
 lastReport=localReport(parsedAssessment);
 el('result').textContent=lastReport;el('saveBtn').disabled=false;
 busy(true);
 msg('التقرير الفوري جاهز للحفظ. جاري طلب تحليل Gemini الاختياري، بمحاولتين كحد أقصى و14 ثانية لكل نموذج.');
 let active=true;
 const warning=setTimeout(()=>{
   if(active)msg('Gemini ما زال يتأخر؛ التقرير الفوري موجود ويمكن حفظه. لا تضغط طلبًا جديدًا.',true);
 },32000);
 google.script.run.withSuccessHandler(r=>{
   if(!active)return;
   active=false;clearTimeout(warning);busy(false);
   lastReport+='\n\n======= GEMINI AI ANALYSIS =======\nModel: '+r.model+
     '\nAnswered: '+r.answeredQuestions+'/'+r.assessedQuestions+
     '\nReported NO responses: '+r.reportedGaps+
     '\nResearch: '+r.researchStatus+'\n\n'+r.analysis+'\n\n'+r.note;
   el('result').textContent=lastReport;el('saveBtn').disabled=false;
   msg('تم إضافة التحليل الذكي إلى التقرير الفوري. لم تتغير درجات التقييم.');
 }).withFailureHandler(e=>{
   if(!active)return;
   active=false;clearTimeout(warning);busy(false);
   msg('تعذر Gemini: '+(e&&e.message?e.message:String(e))+' — التقرير الفوري جاهز ولم يتأثر.',true);
 }).analyzeScaAssessment(assessmentText);
};
el('saveBtn').onclick=()=>{
 const blob=new Blob(['\uFEFF'+lastReport],{type:'text/plain;charset=utf-8'});
 const url=URL.createObjectURL(blob),a=document.createElement('a');
 a.href=url;a.download='SCA_Gemini_Advisory_Brief.txt';a.click();
 setTimeout(()=>URL.revokeObjectURL(url),1000);
};
init();
